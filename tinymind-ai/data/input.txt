Code: create levels in pyRevit

from Autodesk.Revit.DB import Level, Transaction

doc = __revit__.ActiveUIDocument.Document
elevations = [0.0, 3.5, 7.0, 10.5]

transaction = Transaction(doc, "Create levels")
transaction.Start()
try:
    for elevation in elevations:
        Level.Create(doc, elevation)
    transaction.Commit()
except Exception:
    transaction.RollBack()
    raise

from Autodesk.Revit.DB import Level, Transaction


def create_levels(doc, elevations):
    transaction = Transaction(doc, "Create levels")
    transaction.Start()
    try:
        for elevation in elevations:
            Level.Create(doc, elevation)
        transaction.Commit()
    except Exception:
        transaction.RollBack()
        raise


doc = __revit__.ActiveUIDocument.Document
create_levels(doc, [0.0, 3.5, 7.0, 10.5])

    "JOIN_STYLE": 0,
    "DISSOLVE": False,
